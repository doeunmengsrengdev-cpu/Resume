// Production React + TypeScript Component
export function DeployCard({ cluster }: { cluster: ClusterMetrics }) {
  const { data } = useStream(`/api/v1/clusters/${cluster.id}`);

  return (
    <Card className="rounded-2xl p-5 border-zinc-800">
      <StatusHeader region={data.region} uptime={data.uptime} />
      <MetricGrid
        p95Latency={`${data.p95}ms`}
        containers={`${data.active}/${data.total}`}
        throughput={`${data.rps}k/s`}
      />
    </Card>
  );
}
// Edge-cached Node/Next.js API Handler
export async function GET(req: Request) {
  const telemetry = await dockerEngine.inspectCluster({
    region: 'us-east-1',
    includeLatencyPercentiles: true,
  });

  return Response.json({
    uptime: 99.98,
    p95: telemetry.p95Ms,
    active: telemetry.healthyPods,
    rps: telemetry.requestsPerSec,
  }, { headers: { 'Cache-Control': 's-maxage=5' } });
}
